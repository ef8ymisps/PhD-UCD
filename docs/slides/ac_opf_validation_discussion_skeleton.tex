\begin{frame}{Solver Status: What ALMOST\_LOCALLY\_SOLVED Means}
\small
\begin{itemize}
  \item \code{LOCALLY\_SOLVED}: solver reports a local NLP solution satisfying the requested convergence checks.
  \item \code{ALMOST\_LOCALLY\_SOLVED}: solver did not fully satisfy the strict desired termination checks, but reached an acceptable local solution under relaxed/acceptable tolerances.
  \item It is not evidence of global optimality, and it is not automatically a bad result.
  \item For validation, we therefore inspect:
    \begin{itemize}
      \item raw solver status and primal/dual status;
      \item Ipopt original primal infeasibility \code{inf\_pr};
      \item scaled dual infeasibility \code{inf\_du};
      \item barrier parameter \code{mu};
      \item iteration count and termination class.
    \end{itemize}
  \item In the Julia objective-mismatch diagnostic, visible Ipopt checks pass for the tested rows.
\end{itemize}

\begin{frame}{Bus Shunts In The Current Formulation}
\small
\begin{itemize}
  \item In the current bus-injection formulation, bus shunts enter through the diagonal of the admittance matrix.
  \item The diagonal term is built as:
\end{itemize}
\[
Y_{ii}
=
\sum_{\ell \in \delta(i)} y_{\ell,i}
+ g_i^{sh}
+ j b_i^{sh}.
\]
\begin{itemize}
  \item Therefore the diagonal contribution to nodal injections is:
\end{itemize}
\[
P_{ii}=G_{ii} V_i^2,
\qquad
Q_{ii}=-B_{ii} V_i^2.
\]
\begin{itemize}
  \item Since \(G_{ii}\) and \(B_{ii}\) contain \(g_i^{sh}\) and \(b_i^{sh}\), the bus-shunt terms are already present in the current power balances.
  \item Parallel branches are accumulated in \(Y_{ij}\), while branch limits remain per physical branch.
\end{itemize}

\begin{frame}{Current Model vs PM-like Model}
\small
\begin{itemize}
  \item Current Julia model:
    \begin{itemize}
      \item bus-injection balance written directly from sparse \(Y_{\mathrm{bus}}\);
      \item branch flows are nonlinear expressions used for thermal limits and reporting;
      \item shunts are embedded in \(Y_{\mathrm{bus}}\).
    \end{itemize}
  \item PM-like Julia model:
    \begin{itemize}
      \item explicit directed branch-flow variables \(p_{ij}, q_{ij}\);
      \item KCL written as sum of incident arc flows;
      \item bus shunts written explicitly in the balance:
    \end{itemize}
\end{itemize}
\[
\sum p_{ij}
=
\sum P_g - P_d - g_i^{sh} V_i^2,
\qquad
\sum q_{ij}
=
\sum Q_g - Q_d + b_i^{sh} V_i^2.
\]
\begin{itemize}
  \item PM-like was introduced to align the algebraic structure more closely with PowerModels ACP.
\end{itemize}

\begin{frame}{Objective Mismatches: How I Diagnose Them}
\small
\begin{itemize}
  \item First question: is the cheaper solution actually feasible?
  \item Second question: is the PowerModels point feasible in my formulation?
  \item Third question: are equality constraints or inequality levels responsible for the mismatch?
  \item Diagnostic sequence:
    \begin{enumerate}
      \item check solver status and residuals;
      \item compare binding constraints and margins;
      \item run constrained deviation minimization toward PowerModels;
      \item run equality-only deviation minimization toward PowerModels;
      \item evaluate \(h(x,u)\) externally at the projected point;
      \item run PowerModels Newton PF from both OPF endpoints.
    \end{enumerate}
  \item The aim is to distinguish formulation error from local-solver trajectory behavior.
\end{itemize}

\begin{frame}{Constrained Deviation Minimization}
\small
\begin{itemize}
  \item Diagnostic problem:
\end{itemize}
\[
\min
\left\lVert P_g-P_g^{PM}\right\rVert_1
+ \left\lVert Q_g-Q_g^{PM}\right\rVert_1
+ \left\lVert V-V^{PM}\right\rVert_1
+ \left\lVert \theta-\theta^{PM}\right\rVert_1
\]
\begin{itemize}
  \item Subject to the full own AC OPF constraints, including inequalities.
  \item Current model full variant:
    \begin{itemize}
      \item reaches the PowerModels point for the tested objective-mismatch rows;
      \item zero binding mismatches in the full variant.
    \end{itemize}
  \item PM-like model full variant:
    \begin{itemize}
      \item reaches the PowerModels point for tested PM-like mismatch rows;
      \item zero binding mismatches in the full variant.
    \end{itemize}
  \item Ablations help identify which terms matter, but the full formulation can represent the PowerModels point.
\end{itemize}

\begin{frame}{Equality-only Deviation And External \(h(\cdot)\) Levels}
\small
\begin{itemize}
  \item Diagnostic problem:
\end{itemize}
\[
\min
\left\lVert x-x^{PM}\right\rVert_1
+ \left\lVert u-u^{PM}\right\rVert_1
\quad
\text{s.t. only } g(x,u)=0.
\]
\begin{itemize}
  \item Inequalities \(h(x,u)\leq 0\) are not imposed in the solve.
  \item After solving, inequality margins are evaluated externally:
    \begin{itemize}
      \item generator \(P/Q\) bounds;
      \item voltage bounds;
      \item branch thermal limits;
      \item branch angle limits.
    \end{itemize}
  \item Julia finding:
    \begin{itemize}
      \item current and PM-like equality-only diagnostics reach the PowerModels point;
      \item zero equality-only violations;
      \item zero binding mismatches against PowerModels.
    \end{itemize}
\end{itemize}

\begin{frame}{Load-flow Validation With PowerModels Newton PF}
\small

  \item Corrected scheme:
    \begin{enumerate}
      \item write own OPF endpoint into PowerModels data;
      \item write PowerModels OPF endpoint into PowerModels data;
      \item run \code{PowerModels.compute\_ac\_pf} with Newton and \code{flat\_start=false};

      \item compare post-PF inequality levels.
    \end{enumerate}
  \item Finding:
    \begin{itemize}
      \item all 10 endpoint PF runs converge;
      \item branch flows are recalculated in all cases;
      \item same PF violation appears for both endpoints: \(qg_{\min}\), generator 12;
